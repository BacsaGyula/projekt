<script>

const EVENT_END = new Date("2026-10-19T13:00:00+02:00").getTime();

const MAX_POINTS = 2000;

let points = Number(localStorage.getItem("wt_points")) || 0;

const pointsElement = document.getElementById("points");
const maxPointsElement = document.getElementById("maxPoints");
const progressBar = document.getElementById("progressBar");
const percentElement = document.getElementById("percent");
const remainingElement = document.getElementById("remaining");
const stageElement = document.getElementById("stageCurrent");

maxPointsElement.textContent = MAX_POINTS;


function updateProgress() {

    if (points > MAX_POINTS) {
        points = MAX_POINTS;
    }

    if (points < 0) {
        points = 0;
    }

    const percentage =
        Math.round((points / MAX_POINTS) * 100);

    const remaining =
        Math.max(MAX_POINTS - points, 0);

    const stage =
        Math.min(
            Math.floor((points / MAX_POINTS) * 10) + 1,
            10
        );

    pointsElement.textContent = points;
    percentElement.textContent = percentage + "%";
    remainingElement.textContent = remaining;
    stageElement.textContent = stage;

    progressBar.style.width = percentage + "%";

    localStorage.setItem("wt_points", points);

    renderStages();
}


function addPoints() {

    const input =
        document.getElementById("pointInput");

    const value = Number(input.value);

    if (!value || value <= 0) {
        return;
    }

    points += value;

    input.value = "";

    updateProgress();
}


function resetProgress() {

    if (!confirm("Biztosan nullázod az event haladását?")) {
        return;
    }

    points = 0;

    updateProgress();
}


function renderStages() {

    const container =
        document.getElementById("stages");

    container.innerHTML = "";

    const currentStage =
        Math.min(
            Math.floor((points / MAX_POINTS) * 10) + 1,
            10
        );

    for (let i = 1; i <= 10; i++) {

        const stage =
            document.createElement("div");

        stage.className = "stage";

        if (i < currentStage) {
            stage.classList.add("completed");
        }

        if (i === currentStage) {
            stage.classList.add("current");
        }

        stage.innerHTML = `
            <div class="stage-number">
                STAGE ${i}
            </div>

            <div class="stage-icon">
                ${i < currentStage ? "✓" :
                  i === currentStage ? "◆" : "○"}
            </div>
        `;

        container.appendChild(stage);
    }
}


function updateCountdown() {

    const now = Date.now();

    let difference =
        EVENT_END - now;

    if (difference <= 0) {

        document.getElementById("countdown")
            .textContent = "EVENT VÉGET ÉRT";

        return;
    }

    const days =
        Math.floor(difference / 86400000);

    difference %= 86400000;

    const hours =
        Math.floor(difference / 3600000);

    difference %= 3600000;

    const minutes =
        Math.floor(difference / 60000);

    const seconds =
        Math.floor((difference % 60000) / 1000);

    document.getElementById("countdown")
        .textContent =
        `${days}N ${hours}Ó ${minutes}P ${seconds}MP`;
}


document.querySelectorAll(".task input")
.forEach(checkbox => {

    checkbox.addEventListener("change", () => {

        checkbox.parentElement.classList.toggle(
            "done",
            checkbox.checked
        );

    });

});


updateProgress();
updateCountdown();

setInterval(updateCountdown, 1000);

</script>
